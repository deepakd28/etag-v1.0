2. Complete ESP32 Arduino Firmware

        

          Flash this Arduino sketch to your XIAO ESP32-S3. It advertises the matching BLE service, verifies the checksum, and refreshes the Waveshare tri-color display.
        

        
#include <BLEDevice.h>
#include <BLEServer.h>
#include <BLEUtils.h>
#include <BLE2902.h>
#include <GxEPD2_3C.h>

// Matching BLE UUIDs
#define SERVICE_UUID        "e7a10001-5b3c-4f1e-9a21-7c4d8b2f6a10"
#define DATA_CHAR_UUID      "e7a10002-5b3c-4f1e-9a21-7c4d8b2f6a10"
#define STATUS_CHAR_UUID    "e7a10003-5b3c-4f1e-9a21-7c4d8b2f6a10"

// Waveshare 1.54" Tri-Color (200x200) Pins on XIAO ESP32-S3
// CS=D3(4), DC=D2(3), RST=D1(2), BUSY=D0(1)
GxEPD2_3C<GxEPD2_154_Z90c, GxEPD2_154_Z90c::HEIGHT> display(
  GxEPD2_154_Z90c(/*CS=*/ 4, /*DC=*/ 3, /*RST=*/ 2, /*BUSY=*/ 1)
);

const size_t LAYER_SIZE = 5000;
const size_t TOTAL_SIZE = 10000;
uint8_t buffer[TOTAL_SIZE];
size_t expectedLen = 0;
uint8_t expectedChecksum = 0;

BLECharacteristic *pStatusChar = nullptr;

void sendStatus(const char* msg) {
  if (pStatusChar) {
    pStatusChar->setValue(msg);
    pStatusChar->notify();
  }
}

class DataCallbacks : public BLECharacteristicCallbacks {
  void onWrite(BLECharacteristic *pChar) override {
    uint8_t* data = pChar->getData();
    size_t len = pChar->getLength();
    if (len < 1) return;

    uint8_t cmd = data[0];
    if (cmd == 0x01 && len >= 3) { // START
      expectedLen = data[1] | (data[2] << 8);
      memset(buffer, 0, TOTAL_SIZE);
      sendStatus("READY");
    }
    else if (cmd == 0x02 && len > 3) { // DATA CHUNK
      uint16_t chunkIdx = data[1] | (data[2] << 8);
      size_t offset = chunkIdx * 200;
      size_t payloadLen = len - 3;
      if (offset + payloadLen <= TOTAL_SIZE) {
        memcpy(buffer + offset, data + 3, payloadLen);
      }
    }
    else if (cmd == 0x03 && len >= 2) { // END + CHECKSUM
      expectedChecksum = data[1];
      uint8_t sum = 0;
      for (size_t i = 0; i < TOTAL_SIZE; i++) sum = (sum + buffer[i]) & 0xFF;

      if (sum == expectedChecksum) {
        sendStatus("OK");
        sendStatus("REFRESHING");
        // buffer[0..4999] = black layer, buffer[5000..9999] = red layer
        display.init(115200);
        display.setFullWindow();
        display.firstPage();
        do {
          display.drawImage(buffer, buffer + LAYER_SIZE, 0, 0, 200, 200, false, false, true);
        } while (display.nextPage());
        display.powerOff();
        sendStatus("DONE");
      } else {
        sendStatus("ERR:CHECKSUM");
      }
    }
  }
};

void setup() {
  Serial.begin(115200);
  BLEDevice::init("etag");
  BLEServer *pServer = BLEDevice::createServer();
  BLEService *pService = pServer->createService(SERVICE_UUID);

  BLECharacteristic *pDataChar = pService->createCharacteristic(
    DATA_CHAR_UUID,
    BLECharacteristic::PROPERTY_WRITE
  );
  pDataChar->setCallbacks(new DataCallbacks());

  pStatusChar = pService->createCharacteristic(
    STATUS_CHAR_UUID,
    BLECharacteristic::PROPERTY_NOTIFY
  );
  pStatusChar->addDescriptor(new BLE2902());

  pService->start();
  BLEAdvertising *pAdv = BLEDevice::getAdvertising();
  pAdv->addServiceUUID(SERVICE_UUID);
  pAdv->setScanResponse(true);
  BLEDevice::startAdvertising();
}

void loop() {
  delay(100);
}
